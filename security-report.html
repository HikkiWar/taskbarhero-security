TBH-001 · Critical

        
Max HP принимается без серверной валидации

      

    

    

      
Описание
Клиент отправляет maxHp в API, сервер сохраняет его без пересчёта из базовых характеристик. Любой игрок через HTTP-прокси может подставить произвольное значение и стать бессмертным.

      

      
Запрос клиента → Текущая (уязвимая) обработка

        

          

            
❌ Вредоносный HTTP запрос

            
PATCH /v1/user/data HTTP/1.1
Host: game.thebackend.io
Authorization: Bearer eyJ0eXA...

{
  "maxHp": 99999999,
  "currentHp": 99999999,
  "level": 15
}

          

          

            
⚠️ Текущий сервер (принимает)

            
// Уязвимый обработчик:
const data = req.body;
await db.users.update(userId, {
  maxHp:     data.maxHp,     // ← нет проверки!
  currentHp: data.currentHp
});
// HTTP 200 OK
{ "maxHp": 99999999 }  // ← принято!

          

        

      

      
Правильная обработка — BACKND Cloud Function

        

          
✅ Сервер пересчитывает maxHp из базы данных

          
function updatePlayerState(userToken, body) {
  // Шаг 1: Читаем авторитетные данные — клиенту не доверяем
  const stored = Backend.GameInfo.GetV2(userToken, "userStats");

  // Шаг 2: Пересчёт maxHp только из хранилища
  const maxHp = stored.baseHp
    + (stored.level * HP_PER_LEVEL)
    + getEquipBonus(userToken, "hp");  // из инвентаря

  // Шаг 3: currentHp от клиента — но не выше maxHp
  const currentHp = Math.min(body.currentHp ?? maxHp, maxHp);

  // Шаг 4: Сохраняем серверные значения
  Backend.GameInfo.UpdateV2(userToken, "userStats", { maxHp, currentHp });

  // HTTP 200: возвращаем вычисленные значения
  return { maxHp: maxHp, currentHp: currentHp };
  // → { "maxHp": 485, "currentHp": 350 }
}

        

      

    

  


  
  

    

      

      

        
TBH-002 · Critical

        
Alpha/dev эндпоинты зашиты в production-бинарь

      

    

    

      
Описание
В GameAssembly.dll прямо в строках бинаря присутствуют alpha и beta URL BACKND. Эти окружения менее защищены и могут содержать отладочные эндпоинты.

      
Запрос на alpha окружение

        

          

            
❌ Запрос на dev-сервер (из бинаря)

Authorization: Bearer <любой токен>

{ "items": ["all_legendary"] }

// Alpha может вернуть 200 OK
// где prod вернул бы 404

#else
  return "https://game.notification.thebackend.io";
#endif
}
// В Release-сборке alpha-URL НЕ компилируется
// в бинарь — строка физически отсутствует

TBH-007 · Medium

        
Anti-Cheat Toolkit обходится на уровне API

      

    

    

      
Описание
ACTk v6.8.0 шифрует значения в памяти (ObscuredInt, ObscuredFloat). Но когда значение передаётся в HTTP-запрос — оно расшифровано. Burp Suite/mitmproxy видит и меняет его.

      
Схема обхода / Исправление

        

          

            
❌ ACTk не защищает API трафик

            
// В памяти (защищено ACTk):
ObscuredInt maxHp = 485;  // XOR-обфускация

// В момент API запроса (расшифровано):
var data = { maxHp: maxHp.Value }; // = 485

// Burp Suite перехватывает и меняет:
// { "maxHp": 99999999 }
// ACTk ничего об этом не знает

          

          

            
✅ HMAC-подпись + серверная валидация

            
// Подпись запроса:
string Sign(string json, string key) {
  var nonce = Guid.NewGuid().ToString("N")[..8];
  var ts = DateTimeOffset.UtcNow.ToUnixTimeSeconds();
  using var h = new HMACSHA256(Encoding.UTF8.GetBytes(key));
  var sig = Convert.ToBase64String(
    h.ComputeHash(Encoding.UTF8.GetBytes(json+nonce+ts)));
  return $"?nonce={nonce}&ts={ts}&sig={sig}";
}
// Сервер отклоняет запрос с неверной подписью

          

        

      

    

  


  
  

    

      

      

        
TBH-008 · Medium

        
InventoryReceivePostItemAllRequest без серверной проверки

      

    

    

      
Описание
Классы InventoryReceivePostItemAllRequest / InventoryReceivePostItemAllResult найдены в IL2CPP metadata. Клиент постит список предметов напрямую. Сервер принимает их без верификации контекста получения.

      
Запрос / Исправление

        

          

            
❌ Клиент сам задаёт предметы

            
POST /v1/inventory/items HTTP/1.1
Authorization: Bearer <token>

{
  "items": [
    { "itemId": "WPN_KNT_LEG_001", "qty": 99 },
    { "itemId": "GEM_PACK_500", "qty": 99 },
    { "itemId": "ARM_ALL_LEG_001", "qty": 99 }
  ]
}
// HTTP 200 — предметы добавлены!

          

          

            
✅ Клиент сообщает о событии — сервер решает

            
// Клиент отправляет только действие:
POST /v1/actions/monster-kill
{ "monsterId": "boss_01", "zoneId": "dungeon_3",
  "eventId": "a3f-8bc-..." }

// Сервер проверяет бой, роллит дроп:
// → { "granted": [{"itemId": "WPN_KNT_COM_002", "qty": 1}] }
// Клиент не может выбрать predmet

          

        

      

      
Как должен работать сервер

        

          
✅ Cloud Function — сервер авторитарно выдаёт предметы

          
function onMonsterKill(token, monsterId, zoneId, eventId) {
  // 1. Idempotency: уже обработано?
  if (Backend.Cache.Exists(`kill:${eventId}`))
    throw "Already processed";

  // 2. Проверяем, что монстр был в активном бою игрока
  const battle = Backend.GameInfo.GetV2(token, "activeBattle");
  if (!battle?.enemies?.includes(monsterId))
    throw "Monster not in player's battle";

  // 3. Сервер роллит дроп по таблице монстра
  const tableKey = MONSTER_LOOT_MAP[monsterId];
  const drops = Backend.Probability.Roll(tableKey);

  // 4. Сервер напрямую добавляет в инвентарь
  Backend.Inventory.Insert(token, drops.itemId, drops.qty);

  // 5. Помечаем событие обработанным (1 час)
  Backend.Cache.Set(`kill:${eventId}`, 1, ttl: 3600);
  return { granted: drops };
}

static string SafeKey(string steamId, string name) {
  const string SALT = "TBH_2026";
  using var sha = SHA256.Create();
  var h = sha.ComputeHash(
    Encoding.UTF8.GetBytes(steamId + SALT));
  var s = BitConverter.ToString(h, 0, 4).Replace("-","");
  return $"{name}_{s}";
}
// → "TermsOfServiceVersion_A3F1C209"
// Steam ID не раскрыт

          

        

      

    

  


  
  

    

      

      

        
TBH-012 · High

        
Характеристики снаряжения подменяются клиентом (Equipment Stat Injection)

      

    

    

      
Описание
Когда игрок надевает предмет, клиент отправляет не только itemId, но и характеристики: atk, def, hp. Если сервер использует эти значения напрямую для боевых расчётов — вместо того чтобы смотреть реальные статы предмета в собственной БД — атакующий меняет "atk": 12 на "atk": 99999 и мгновенно получает урон в 99k+. Это не атака на сервер — это атака на то, чему сервер доверяет.

      
Атака — снаряжение Iron Sword

        

          

            
❌ Клиент подставляет статы снаряжения

            
POST /v1/equip HTTP/1.1
Authorization: Bearer <token>

{
  "slot": "weapon",
  "itemId": "WPN_KNT_COM_001",
  // Реальные статы: atk:12, def:0, hp:0
  "stats": {
    "atk": 99999,   // ← подменено!
    "def": 9999,
    "hp":  99999
  }
}
// Сервер сохраняет "atk": 99999
// Урон в бою = 99999

          

          

            
✅ Сервер смотрит БД, игнорирует stats

            
function equipItem(token, slot, itemId) {
  // Принимаем только itemId и slot
  // stats от клиента — ИГНОРИРУЮТСЯ

  // Сервер смотрит СВОЮ БД предметов:
  const def = ITEM_DB[itemId];
  if(!def) throw "Unknown item";

  // Сохраняем серверные статы:
  Backend.GameInfo.UpdateV2(token, "equip", {
    [slot]: { itemId,
      atk: def.atk,   // = 12 из БД
      def: def.def,   // = 0
      hp:  def.hp     // = 0
    }
  });
  return { slot, itemId, stats: def };
}

          

        

      

      
Авторитетная база данных предметов на сервере (фрагмент)

        

          
✅ ITEM_DB — единственный источник статов

          
// Cloud Function / BACKND Chart Table
const ITEM_DB = {
  "WPN_KNT_COM_001": { atk: 12,  def: 0,  hp: 0,   slot: "weapon", allowedClass: "Knight" },
  "WPN_KNT_RAR_001": { atk: 45,  def: 0,  hp: 0,   slot: "weapon", allowedClass: "Knight" },
  "WPN_KNT_LEG_001": { atk: 160, def: 40, hp: 0,   slot: "weapon", allowedClass: "Knight" },
  "ARM_KNT_RAR_001": { atk: 0,   def: 55, hp: 200, slot: "armor",  allowedClass: "Knight" },
  "WPN_WIZ_LEG_001": { atk: 130, def: 0,  hp: 0,   slot: "weapon", allowedClass: "Wizard" },
  // ... все предметы
};

// Дополнительные проверки при экипировке:
function validateEquip(token, slot, itemId) {
  const def = ITEM_DB[itemId];
  if(!def) throw "Unknown item";
  if(def.slot !== slot) throw "Wrong slot";

  const player = Backend.GameInfo.GetV2(token, "player");
  if(def.allowedClass !== player.class)
    throw "Wrong class"; // рыцарь не носит посох

  const inv = Backend.Inventory.Get(token);
  if(!inv.items[itemId])
    throw "Item not in inventory"; // дюп-защита
}
